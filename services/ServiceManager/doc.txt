// Helper to write string to file
static bool write_file_string(const char* path, const std::string& content, mode_t mode) {
    int fd = open(path, O_WRONLY | O_CREAT | O_TRUNC, mode);
    if (fd < 0) return false;
    ssize_t written = write(fd, content.data(), content.size());
    close(fd);
    return written == static_cast<ssize_t>(content.size());
}

// Ensures /etc/machine-id and /var/lib/dbus/machine-id exist, as dbus-daemon requires a machine ID
void ensure_machine_id() {
    const char* etc_id = "/etc/machine-id";
    const char* dbus_id = "/var/lib/dbus/machine-id";

    struct stat st;
    if (stat(etc_id, &st) == 0 && st.st_size >= 32) {
        return; // Already initialized
    }

    mkdir("/var/lib", 0755);
    mkdir("/var/lib/dbus", 0755);

    char hex_str[33] = {0};
    int fd = open("/dev/urandom", O_RDONLY);
    if (fd >= 0) {
        unsigned char rand_bytes[16];
        if (read(fd, rand_bytes, sizeof(rand_bytes)) == sizeof(rand_bytes)) {
            for (int i = 0; i < 16; ++i) {
                snprintf(&hex_str[i * 2], 3, "%02x", rand_bytes[i]);
            }
        }
        close(fd);
    }

    std::string machine_id = (hex_str[0] != '\0') ? hex_str : "0123456789abcdef0123456789abcdef";
    machine_id += "\n";

    if (write_file_string(etc_id, machine_id, 0644)) {
        log_msg(1, "[INIT] Generated %s\n", etc_id);
    }
    write_file_string(dbus_id, machine_id, 0644);
}

void ensure_wpa_config(const char* conf_path) {
    struct stat st;
    if (stat(conf_path, &st) == 0 && st.st_size > 0) {
        return;
    }

    mkdir("/etc/wpa_supplicant", 0755);

    const std::string default_conf =
        "ctrl_interface=/var/run/wpa_supplicant\n"
        "ctrl_interface_group=0\n"
        "update_config=1\n";

    if (write_file_string(conf_path, default_conf, 0644)) {
        log_msg(1, "[INIT] Created default %s\n", conf_path);
    }
}

pid_t spawn_daemon(const char* name, const std::vector<const char*>& args) {
    pid_t pid = fork();
    if (pid < 0) {
        log_msg(2, "[DAEMON] Failed to fork %s: %s\n", name, std::strerror(errno));
        return -1;
    }
    if (pid == 0) {
        std::vector<char*> argv;
        for (const auto arg : args) {
            argv.push_back(const_cast<char*>(arg));
        }
        argv.push_back(nullptr);

        execvp(argv[0], argv.data());
        log_msg(2, "[DAEMON] Failed to exec %s (%s): %s\n", name, argv[0], std::strerror(errno));
        _exit(127);
    }
    log_msg(1, "[DAEMON] Started %s (PID: %d)\n", name, pid);
    return pid;
}

void start_daemons(pid_t& pid_dbus, pid_t& pid_bt, pid_t& pid_wpa, pid_t& pid_sshd, const char* eth0_ip = "192.168.0.100") {
    setenv("PATH", "/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin:/usr/libexec/bluetooth:/usr/lib/bluetooth", 1);
    setenv("DBUS_SYSTEM_BUS_ADDRESS", "unix:path=/var/run/dbus/system_bus_socket", 1);

    mkdir("/run", 0755);
    mkdir("/run/dbus", 0755);
    mkdir("/var", 0755);
    mkdir("/var/run", 0755);
    mkdir("/var/run/dbus", 0755);

    unlink("/var/run/dbus/system_bus_socket");
    unlink("/run/dbus/system_bus_socket");
    unlink("/var/run/dbus/pid");
    unlink("/run/dbus/pid");

    ensure_machine_id();

    log_msg(1, "[INIT] Starting D-Bus system bus daemon...\n");
    pid_dbus = spawn_daemon("dbus-daemon", {"dbus-daemon", "--system", "--nofork", "--nopidfile"});

    log_msg(1, "[INIT] Waiting for D-Bus socket...\n");
    bool dbus_ready = false;
    for (int i = 0; i < 40; ++i) {
        struct stat st;
        if (stat("/var/run/dbus/system_bus_socket", &st) == 0 ||
            stat("/run/dbus/system_bus_socket", &st) == 0) {
            dbus_ready = true;
            break;
        }
        int status;
        if (waitpid(pid_dbus, &status, WNOHANG) > 0) {
            log_msg(2, "[INIT] D-Bus daemon exited early.\n");
            break;
        }
        usleep(50000);
    }

    if (dbus_ready) {
        log_msg(1, "[INIT] D-Bus system bus is ready.\n");
        chmod("/var/run/dbus/system_bus_socket", 0666);
    } else {
        log_msg(2, "[INIT] Warning: D-Bus socket not detected after timeout.\n");
    }

    log_msg(1, "[INIT] Starting bluetoothd...\n");
    pid_bt = spawn_daemon("bluetoothd", {"bluetoothd", "--nodetach"});

    log_msg(1, "[INIT] Starting wpa_supplicant...\n");
    mkdir("/run/wpa_supplicant", 0755);
    mkdir("/var/run/wpa_supplicant", 0755);
    chmod("/var/run/wpa_supplicant", 0777); 

    const char* wpa_conf = "/etc/wpa_supplicant.conf";
    ensure_wpa_config(wpa_conf);

    std::vector<const char*> wpa_args = {
        "wpa_supplicant",
        "-u",           
        "-s",           
        "-c", wpa_conf  
    };

    struct stat net_st;
    if (stat("/sys/class/net/wlan0", &net_st) == 0) {
        wpa_args.push_back("-i");
        wpa_args.push_back("wlan0");
    } else {
        wpa_args.push_back("-g");
        wpa_args.push_back("/var/run/wpa_supplicant/global");
    }

    pid_wpa = spawn_daemon("wpa_supplicant", wpa_args);

    // 7. Start sshd (restricted to eth0 IP, root without password)
    log_msg(1, "[INIT] Starting sshd (eth0: %s only, root without password)...\n", eth0_ip);
    mkdir("/var/empty", 0700);
    mkdir("/run/sshd", 0755);
    mkdir("/root", 0700);

    ensure_ssh_host_keys();

    std::string listen_opt = std::string("ListenAddress=") + eth0_ip;

    std::vector<const char*> sshd_args = {
        "sshd",
        "-D",                               // Run in foreground for supervision
        "-e",                               // Log to stderr/console
        "-o", "PermitRootLogin=yes",        // Allow root login
        "-o", "PermitEmptyPasswords=yes",   // Allow login with empty password
        "-o", "PasswordAuthentication=yes", // Allow password authentication
        "-o", listen_opt.c_str(),           // Bind only to eth0 IP address
        "-o", "UsePAM=no",                  // Avoid PAM dependency in embedded rootfs
        "-o", "StrictModes=no"              // Avoid strict file permission checks
    };

    pid_sshd = spawn_daemon("sshd", sshd_args);
}

void ensure_ssh_host_keys() {
    mkdir("/etc/ssh", 0755);

    struct stat st;
    if (stat("/etc/ssh/ssh_host_ed25519_key", &st) == 0 ||
        stat("/etc/ssh/ssh_host_rsa_key", &st) == 0) {
        return;
    }

    pid_t pid = fork();
    if (pid == 0) {
        char* const args[] = {
            const_cast<char*>("ssh-keygen"),
            const_cast<char*>("-A"),
            nullptr
        };
        execvp("ssh-keygen", args);

        char* const args2[] = {
            const_cast<char*>("ssh-keygen"),
            const_cast<char*>("-t"),
            const_cast<char*>("ed25519"),
            const_cast<char*>("-f"),
            const_cast<char*>("/etc/ssh/ssh_host_ed25519_key"),
            const_cast<char*>("-N"),
            const_cast<char*>(""),
            nullptr
        };
        execvp("ssh-keygen", args2);
        _exit(1);
    }
    if (pid > 0) {
        // Wait at most 2 seconds for ssh-keygen, never block PID 1 indefinitely on entropy
        for (int i = 0; i < 20; ++i) {
            int status;
            pid_t res = waitpid(pid, &status, WNOHANG);
            if (res != 0) break;
            usleep(100000);
        }
    }
}

// Ensures dropbear host keys exist or are generated
void ensure_dropbear_keys() {
    mkdir("/etc/dropbear", 0755);
    struct stat st;

        return;

    }
    if (pid > 0) {
        for (int i = 0; i < 20; ++i) {
            int status;
            pid_t res = waitpid(pid, &status, WNOHANG);
            if (res != 0) break;
            usleep(100000);
        }
    }
}

// Spawns dropbear bound strictly to eth0 IP, allowing root without password
pid_t start_dropbear(const char* bind_ip) {
    mkdir("/etc/dropbear", 0755);
    mkdir("/var/log", 0755);
    mkdir("/var/run", 0755);
    mkdir("/root", 0700);

    ensure_dropbear_keys();

    std::string bind_arg = std::string(bind_ip) + ":22";

    std::vector<const char*> args = {
        "dropbear",
        "-F",                   // Run in foreground for process supervisor
        "-E",                   // Log output to stderr
        "-p", bind_arg.c_str(), // Bind only to eth0 IP address:port 22
        "-B",                   // Allow blank/empty passwords (root without password)
        "-R"                    // Generate hostkeys automatically if missing
    };

    return spawn_daemon("dropbear", args);
}

pid_t spawn_shell() {
    pid_t pid = fork();
    if (pid < 0) {
        log_msg(2, "[INIT] Error: Fork failed for shell: %s\n", std::strerror(errno));
        return -1;
    }

    if (pid == 0) {
        // --- CHILD PROCESS (SHELL) ---
        // Ignore job-control stopping signals so child process is never suspended
        signal(SIGTTOU, SIG_IGN);
        signal(SIGTTIN, SIG_IGN);
        signal(SIGTSTP, SIG_IGN);

        // 1. Connect stdin (0), stdout (1), stderr (2) to the active serial or console tty
        const char* tty_candidates[] = {
            "/dev/ttyAMA0",
            "/dev/console",
            "/dev/ttyS0",
            "/dev/tty0"
        };

        int console_fd = -1;
        for (const char* dev : tty_candidates) {
            console_fd = open(dev, O_RDWR);
            if (console_fd >= 0) break;
        }

        if (console_fd >= 0) {
            dup2(console_fd, STDIN_FILENO);
            dup2(console_fd, STDOUT_FILENO);
            dup2(console_fd, STDERR_FILENO);
            if (console_fd > STDERR_FILENO) {
                close(console_fd);
            }
        }

        // 2. Claim controlling terminal and session
        setsid();
        ioctl(STDIN_FILENO, TIOCSCTTY, 1);

        // 3. Set terminal line mode (ECHO, canonical, newline translation)
        struct termios tty;
        if (tcgetattr(STDIN_FILENO, &tty) == 0) {
            tty.c_cflag |= (CREAD | CLOCAL);
            tty.c_oflag |= (OPOST | ONLCR);
            tty.c_lflag |= (ISIG | ICANON | ECHO | ECHOE | ECHOK);
            tcsetattr(STDIN_FILENO, TCSANOW, &tty);
        }

        // 4. Clear terminal screen
        if (write(STDOUT_FILENO, "\033[2J\033[H", 7) < 0) { /* ignore */ }

        // 5. Set up environment for root shell
        setenv("USER", "root", 1);
        setenv("LOGNAME", "root", 1);
        setenv("HOME", "/root", 1);
        setenv("TERM", "vt100", 1);
        setenv("PATH", "/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin", 1);

        mkdir("/root", 0700);
        if (chdir("/root") != 0) {
            if (chdir("/") != 0) { /* ignore */ }
        }

        // 6. Detect and execute the available shell binary as a login shell
        const char* shell_candidates[] = {
            "/bin/sh",
            "/bin/bash",
            "/bin/ash",
            "/sbin/sh",
            "/usr/bin/sh"
        };

        for (const char* candidate : shell_candidates) {
            if (access(candidate, X_OK) == 0) {
                char* const args[] = { const_cast<char*>("-sh"), nullptr };
                execv(candidate, args);

                char* const args2[] = { const_cast<char*>("sh"), nullptr };
                execv(candidate, args2);
            }
        }

        // Fallback search in PATH
        char* const fallback_args[] = { const_cast<char*>("-sh"), nullptr };
        execvp("sh", fallback_args);

        const char* err_msg = "Error: Failed to start shell on console\n";
        if (write(STDERR_FILENO, err_msg, std::strlen(err_msg)) < 0) { /* ignore */ }
        _exit(1);
    }

    log_msg(1, "[INIT] Started interactive shell on console (PID: %d)\n", pid);
    return pid;
}
